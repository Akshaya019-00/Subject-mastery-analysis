DAX Quick Reference · Key Measures & Calculated Columns
Calc ColGrade
SWITCH(TRUE(), [TotalMarks]>=90,"O", [TotalMarks]>=80,"A+", [TotalMarks]>=70,"A", [TotalMarks]>=60,"B+", [TotalMarks]>=50,"B", [TotalMarks]>=40,"C","F")
Calc ColRisk_Category
IF([TotalMarks] < 35, "At-Risk", IF([TotalMarks] < 50, "Borderline", "Safe" ) )
MeasurePass_Percentage
DIVIDE( COUNTROWS(FILTER( Fact_Marks, Fact_Marks[TotalMarks]>=40 )), COUNTROWS(Fact_Marks), 0 ) * 100
MeasureCount_AtRisk
CALCULATE( DISTINCTCOUNT( Fact_Marks[StudentID] ), Fact_Marks[Risk_Category] = "At-Risk" )
MeasureStudent_Rank
RANKX( ALL( Dim_Student[StudentName] ), [Avg_Marks], , DESC, DENSE )
MeasureSubject_Avg
CALCULATE( AVERAGE( Fact_Marks[TotalMarks] ), ALLEXCEPT( Fact_Marks, Dim_Subject[SubjectName] ) )
